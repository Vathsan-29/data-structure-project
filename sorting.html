Sorting student records

Selection, insertion and radix sort

Data structureArrays
Syllabus unitUnit I
Menu option4

Student pointers are copied into an array and sorted by selection sort (by ID), insertion sort (by name, ignoring case) or radix sort (by ID, one pass per digit).

7.9 Sort Student Records (Option 4)

static void print_sorted(Student *a[], int n)
{
    int i;
    line('-', 50);
    printf("%-12s %-20s %-4s\n", "ID", "NAME", "YEAR");
    line('-', 50);
    for (i = 0; i < n; i++)
        printf("%-12d %-20s %-4d\n", a[i]->id, a[i]->name, a[i]->year);
    line('-', 50);
}

static void sort_students(void)
{
    Student *a[MAX_STUDENTS], *t, *tmp;
    int n = 0, i, j, min, choice;

    if (head == NULL) {
        printf("No students registered yet.\n");
        return;
    }
    for (t = head; t != NULL; t = t->next)
        a[n++] = t;





    printf("\n 1. Selection sort by ID\n");
    printf(" 2. Insertion sort by name\n");
    printf(" 3. Radix sort by ID\n");
    choice = read_int("Enter your choice: ");

    if (choice == 1) {
        for (i = 0; i < n - 1; i++) {
            min = i;
            for (j = i + 1; j < n; j++)
                if (a[j]->id < a[min]->id)
                     min = j;
            tmp = a[i];
            a[i] = a[min];
            a[min] = tmp;
        }
        printf("Sorted by ID (selection sort):\n");
        print_sorted(a, n);
    } else if (choice == 2) {
        for (i = 1; i < n; i++) {
            tmp = a[i];
            j = i - 1;
            while (j >= 0 && name_compare(a[j]->name, tmp->name) > 0) {
                a[j + 1] = a[j];
                j--;
            }
            a[j + 1] = tmp;
        }
        printf("Sorted by name (insertion sort):\n");
        print_sorted(a, n);
    } else if (choice == 3) {
        Student *out[MAX_STUDENTS];
        int count[10];
        long long exp;
        int max = a[0]->id, d;

        for (i = 1; i < n; i++)
            if (a[i]->id > max)
                max = a[i]->id;

        for (exp = 1; max / exp > 0; exp *= 10) {    /* one pass per digit */
             for (d = 0; d < 10; d++)
                 count[d] = 0;
             for (i = 0; i < n; i++)
                 count[(a[i]->id / exp) % 10]++;
             for (d = 1; d < 10; d++)
                 count[d] += count[d - 1];
             for (i = n - 1; i >= 0; i--) {
                 d = (int)((a[i]->id / exp) % 10);
                 out[--count[d]] = a[i];
             }
             for (i = 0; i < n; i++)
                 a[i] = out[i];
        }
        printf("Sorted by ID (radix sort):\n");
        print_sorted(a, n);
    } else {
        printf("Invalid choice.\n");
    }
}
